【算法】

            不难发现，题目中所说的编码方式就是哈夫曼编码

            注意合并时优先合并深度小的

【代码】

            

#include<bits/stdc++.h>
using namespace std;
#define MAXN 100010

struct info
{
        long long w,s;
        friend bool operator < (info a,info b)
        {
                return (a.w != b.w) ? (a.w > b.w) : (a.s > b.s);
        }
};

int i;
long long w[MAXN];
info tmp;
priority_queue< info > q;
long long ans1,val,mx,n,k,ans2;

int main() 
{
        
        scanf("%lld%lld",&n,&k);
        for (i = 1; i <= n; i++) 
        {
                scanf("%lld",&w[i]);
                q.push((info){w[i],0});
        }
        while ((n - 1) % (k - 1) != 0)
        {
                q.push((info){0,0});
                n++;
        }
        while (q.size() >= k)
        {
                val = 0; mx = 0;
                for (i = 1; i <= k; i++)
                {
                        tmp = q.top();
                        q.pop();
                        val += tmp.w;    
                        mx = max(mx,tmp.s);
                }        
                ans1 += val;
                ans2 = max(ans2,mx+1);
                q.push((info){val,mx+1});
        }
        printf("%lld\n%lld\n",ans1,ans2);
        
        return 0;
    
}
