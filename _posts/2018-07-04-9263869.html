【算法】

           搜索剪枝

           剪枝1 ： 优化搜索顺序，从大到小枚举

           剪枝2 ： Ai + Aj可能相等，只需搜一次即可

           剪枝3 ： 通过观察发现 : m <= 10,可以用迭代加深搜索

using namespace std;

int i,n,step;
int ans[15];

inline bool dfs(int dep)
{
        int i,j;
        bool visited[110];
        if (dep > step)
        {
                if (ans[step] == n) return true;
                else return false;        
        }    else
        {
                memset(visited,false,sizeof(visited));
                for (i = dep - 1; i >= 1; i--)
                {
                        for (j = i; j >= 1; j--)
                        {
                                if (ans[i] + ans[j] <= ans[i-1]) break;
                                if (ans[i] + ans[j] > n || visited[ans[i]+ans[j]]) continue;
                                visited[ans[i]+ans[j]] = true;
                                ans[dep] = ans[i] + ans[j];
                                if (dfs(dep+1)) return true;    
                        }    
                }
                return false;
        }
}

int main() 
{
        
        while (scanf("%d",&n) != EOF && n)
        {
                ans[1] = 1;
                for (i = 1; i <= 10; i++)
                {
                        step = i;
                        if (dfs(2)) break;
                }
                for (i = 1; i <= step; i++) printf("%d ",ans[i]);
                printf("\n");
        }
        
        return 0;
    
}
