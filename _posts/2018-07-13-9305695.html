using namespace std;
#define MAXN 50010

struct info
{
        int x,y;
} a[MAXN];

int i,n;
int ans[MAXN];

struct BinaryIndexedTree
{
        int c[MAXN];
        inline int lowbit(int x)
        {
                return x & (-x);        
        }        
        inline void modify(int pos,int val)
        {
                int i;
                for (i = pos; i < MAXN; i += lowbit(i)) c[i] += val; 
        }
        inline int query(int pos)
        {
                int i;
                int ret = 0;
                for (i = pos; i; i -= lowbit(i))
                        ret += c[i];
                return ret;
        }
} BIT;

int main() 
{
        
        scanf("%d",&n);
        for (i = 1; i <= n; i++)
        {
                scanf("%d%d",&a[i].x,&a[i].y);
                a[i].x++; a[i].y++;        
        }
        for (i = 1; i <= n; i++)
        {
                ans[BIT.query(a[i].x)]++;
                BIT.modify(a[i].x,1);        
        }
        for (i = 0; i < n; i++) printf("%d\n",ans[i]);
        
        return 0;
    
}
