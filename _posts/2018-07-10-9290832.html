【算法】

           列出异或方程组，用高斯消元求解即可

【代码】

          

#include <algorithm>

using namespace std;

int i,j,k,T,TC,cnt;
int a[50],f[50],pos[50];

int main() 
{
        
        scanf("%d",&T);
        while (T--)
        {
                memset(f,true,sizeof(f));
                cnt = 0;
                memset(a,0,sizeof(a));
                for (i = 1; i <= 30; i++) scanf("%d",&a[i]);        
                for (i = 1; i <= 5; i++)
                {
                        for (j = 1; j <= 6; j++)    
                        {
                                if (i != 1) a[(i-1)*6+j] |= 1 << ((i - 2) * 6 + j);
                                if (i != 5) a[(i-1)*6+j] |=  1 << (i * 6 + j);
                                if (j != 1) a[(i-1)*6+j] |= 1 << ((i - 1) * 6 + j - 1);
                                if (j != 6) a[(i-1)*6+j] |= 1 << ((i - 1) * 6 + j + 1);
                                a[(i-1)*6+j] |= 1 << ((i - 1) * 6 + j);
                        }
                }        
                for (i = 1; i <= 30; i++)
                {
                        for (j = i + 1; j <= 30; j++)
                        {
                                if (a[j] > a[i])
                                        swap(a[i],a[j]);
                        }
                        if (a[i] == 0) break;
                        for (k = 30; k; k--)
                        {
                                if (a[i] & (1 << k))
                                {
                                        for (j = 1; j <= 30; j++)
                                        {
                                                if (i != j && (a[j] & (1 << k)))
                                                        a[j] ^= a[i];
                                        }
                                        f[k] = false;
                                        pos[k] = i;
                                        break;
                                }
                        }
                }
                printf("PUZZLE #%d\n",++TC);
                for (i = 1; i <= 30; i++)
                {
                        if (f[i]) printf("0 ");
                        else printf("%d ",a[pos[i]]&1);
                        if (i % 6 == 0) printf("\n");
                }
        }    
        
        
        return 0;
    
}
