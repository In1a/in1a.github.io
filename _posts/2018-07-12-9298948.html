【算法】

             n为偶数时必胜，否则必败

【代码】

          

#include<bits/stdc++.h>
using namespace std;

int n;

int main() 
{
        
        while (scanf("%d",&n) && n)
        {
                if (n & 1) printf("Bob\n");
                else printf("Alice\n");
        }
        
        return 0;
    
}
