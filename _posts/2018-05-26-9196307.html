using namespace std;
#define MAXN 910

int i,n,m,u,v,s,root,tot;
int head[MAXN],fa[MAXN],f[MAXN],ans[MAXN];
bool visited[MAXN];
vector<int> query[MAXN];

struct Edge
{
        int to,nxt;
} e[MAXN];

inline int find(int x)
{
        if (f[x] == x) return x;
        return f[x] = find(f[x]);
}
inline void tarjan(int u)
{
        int i,v;
        visited[u] = true;
        f[u] = u;
        for (i = 0; i < query[u].size(); i++)
        {
                v = query[u][i];
                if (visited[v]) ans[find(v)]++;
        }        
        for (i = head[u]; i; i = e[i].nxt)
        {
                v = e[i].to;
                tarjan(v);
                f[v] = u;
        }
}
inline void add(int u,int v)
{
        tot++;
        e[tot] = (Edge){v,head[u]};
        head[u] = tot;
}

int main() 
{
        
        while (scanf("%d",&n) != EOF)
        {
                tot = 0;
                for (i = 1; i <= n; i++) 
                {
                        head[i] = 0;
                        fa[i] = 0;
                        ans[i] = 0;
                        visited[i] = false;
                        query[i].clear();
                }
                for (i = 1; i <= n; i++)
                {
                        scanf("%d : (%d)",&u,&s);
                        while (s--)
                        {
                                scanf("%d",&v);
                                add(u,v);
                                fa[v] = u;    
                        }    
                }    
                for (i = 1; i <= n; i++) 
                {
                        if (!fa[i]) 
                                root = i;
                }
                scanf("%d",&m);
                for (i = 1; i <= m; i++)
                {
                        scanf(" (%d %d)",&u,&v);
                        query[u].push_back(v);
                        query[v].push_back(u);
                }
                tarjan(root);
                for (i = 1; i <= n; i++)
                {
                        if (ans[i])
                                printf("%d:%d\n",i,ans[i]);
                }
        }
        
        return 0;
    
}
