using namespace std;
#define MAXN 110
#define MAXM 100010

int i,j,n,m,cnt;
int a[MAXN],c[MAXN],used[MAXM];
bool f[MAXM];

int main()
{
    
    while (scanf("%d%d",&n,&m) && (n || m))
    {
        memset(f,false,sizeof(f));
        for (i = 1; i <= n; i++) scanf("%d",&a[i]);
        for (i = 1; i <= n; i++) scanf("%d",&c[i]);
        f[0] = true;
        for (i = 1; i <= n; i++)
        {
            for (j = 0; j <= m; j++) used[j] = 0;
            for (j = a[i]; j <= m; j++)
            {
                if (!f[j] && f[j - a[i]] && used[j - a[i]] < c[i])
                {
                    f[j] = true;
                    used[j] = used[j - a[i]] + 1;
                }
            }
        }
        cnt = 0;
        for (i = 1; i <= m; i++)
        {
            if (f[i])
                cnt++;
        }
        printf("%d\n",cnt);
    }

    return 0;
}
