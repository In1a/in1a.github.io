【题目链接】

           http://codeforces.com/contest/451/problem/B

【算法】

            模拟

            在序列中找到一段单调递增的子序列，将这段序列反转，然后判断序列是否变得单调递增，即可

【代码】

           

#include<bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 10;

int i,n,l,r;
bool flag;
int a[MAXN];

int main() 
{
        
        scanf("%d",&n);
        for (i = 1; i <= n; i++) scanf("%d",&a[i]);
        l = r = 1;
        flag = false;
        for (i = 2; i <= n; i++)
        {
                if (a[i] < a[i-1]) 
                {
                        if (!flag) 
                        {
                                l = i - 1;
                                r = i;
                        }
                        else r++;    
                        flag = true;
                } else if (flag) break; 
        }
        reverse(a+l,a+r+1);
        for (i = 1; i <= n; i++)
        {
                if (a[i] < a[i-1])
                {
                        printf("no\n");
                        return 0;
                } 
        }
        printf("yes\n");
        printf("%d %d\n",l,r);
         
        return 0;
    
}
