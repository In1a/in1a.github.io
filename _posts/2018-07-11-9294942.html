using namespace std;
#define MAXN 1010

int i,j,n,s;
double f[MAXN][MAXN];

int main() 
{
    
        while (scanf("%d%d",&n,&s) != EOF)
        {
                memset(f,0,sizeof(f));
                f[n][s] = 0;
                for (i = n; i >= 0; i--)
                {
                        for (j = s; j >= 0; j--)
                        {
                                if (i == n && j == s) continue;
                                f[i][j] = (1.0 * i * (s - j) * f[i][j+1] + 1.0 * (n - i) * j * f[i+1][j] + 1.0 * (n - i) * (s - j) * f[i+1][j+1] + n * s) / (n * s - i * j);
                        }
                }
                printf("%.4lf\n",f[0][0]);
        }
    
        return 0;
    
}
