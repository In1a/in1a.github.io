【算法】

           优先队列BFS

           实现类似于堆优化dijkstra

using namespace std;
#define MAXN 1010
#define MAXM 10010
#define MAXC 110
const int INF = 2e9;

int i,n,m,q,tot,u,v,d,c,s,t,ans;
int head[MAXN];
bool visited[MAXN][MAXC];
int dist[MAXN][MAXC];
int p[MAXN];

struct info
{
        int pos,w,d;
        friend bool operator < (info a,info b)
        {
                return a.d > b.d;
        }
} ;
struct Edge
{
        int to,w,nxt;
} e[MAXM<<1];

inline void add(int u,int v,int w)
{
        tot++;
        e[tot] = (Edge){v,w,head[u]};
        head[u] = tot;
}
inline int bfs(int c,int s,int t)
{
        int i,j;
        priority_queue< info > q;
        info cur;
        for (i = 0; i <= n; i++)
        {
                for (j = 0; j <= c; j++)
                {
                        dist[i][j] = INF;
                }
        }
        memset(visited,false,sizeof(visited));
        dist[s][0] = 0;
        while (!q.empty()) q.pop(); 
        q.push((info){s,0,0});
        while (!q.empty())
        {
                cur = q.top();
                q.pop();
                if (visited[cur.pos][cur.w]) continue;
                if (cur.pos == t) return dist[cur.pos][cur.w];
                visited[cur.pos][cur.w] = true;
                if (cur.w < c)
                {
                        if (dist[cur.pos][cur.w] + p[cur.pos] < dist[cur.pos][cur.w+1])
                        {
                                dist[cur.pos][cur.w+1] = dist[cur.pos][cur.w] + p[cur.pos];
                                q.push((info){cur.pos,cur.w+1,dist[cur.pos][cur.w+1]});
                        }
                }
                for (i = head[cur.pos]; i; i = e[i].nxt)
                {
                        if (cur.w >= e[i].w) 
                        {
                                if (dist[cur.pos][cur.w] < dist[e[i].to][cur.w-e[i].w])
                                {
                                        dist[e[i].to][cur.w-e[i].w] = dist[cur.pos][cur.w];
                                        q.push((info){e[i].to,cur.w-e[i].w,dist[e[i].to][cur.w-e[i].w]});
                                }
                        }
                }    
        }    
        return -1;
}

int main() 
{
        
        scanf("%d%d",&n,&m);
        for (i = 0; i < n; i++) scanf("%d",&p[i]);
        for (i = 1; i <= m; i++)
        {
                scanf("%d%d%d",&u,&v,&d);
                add(u,v,d); 
                add(v,u,d);        
        }
        scanf("%d",&q);
        while (q--)
        {
                scanf("%d%d%d",&c,&s,&t);
                ans = bfs(c,s,t);
                if (ans == -1) printf("impossible\n");
                else printf("%d\n",ans);        
        }
        
        return 0;
    
}
