#include<bits/stdc++.h>
using namespace std;
#define MAXN 1010
#define MAXL 10000

struct info { 
        int l,r; 
} a[MAXN];
int i,n;
struct INT {
        int len;
        int num[MAXL];    
} sum,ans,val;

template <typename T> inline void read(T &x) {
        int f = 1; x = 0;
        char c = getchar();
        for (; !isdigit(c); c = getchar()) { if (c == '-') f = -f; }
        for (; isdigit(c); c = getchar()) x = (x << 3) + (x << 1) + c - '0';
        x *= f;
}
template <typename T> inline void write(T x) {
    if (x < 0) { putchar('-'); x = -x; }
    if (x > 9) write(x/10);
    putchar(x%10+'0');
}
template <typename T> inline void writeln(T x) {
    write(x);
    puts("");
}
inline bool cmp(info a,info b) { return a.l * a.r < b.l * b.r; }
inline void multipy(INT &x,int y) {
        int i;
        for (i = 0; i < x.len; i++) x.num[i] *= y;
        for (i = 0; i < x.len; i++) {
                if (x.num[i] >= 10) {
                        x.num[i+1] += x.num[i] / 10;
                        x.num[i] %= 10;
                }
        }
        while (x.num[x.len]) {
                if (x.num[x.len] >= 10) {
                        x.num[x.len+1] += x.num[x.len] / 10;
                        x.num[x.len] %= 10; 
                }
                x.len++;
        }
}
inline INT divide(INT x,int y) {
        int i,t=0;
        static INT ret;
        ret.len = 0;
        for (i = x.len - 1; i >= 0; i--) {
                t = t * 10 + x.num[i]; 
                ret.num[ret.len++] = t / y;
                t %= y;
        }
        reverse(ret.num,ret.num+ret.len);
        while (ret.num[ret.len-1] == 0 && ret.len > 1) ret.len--;
        return ret;
}
inline bool comp(INT x,INT y) {
        int i;
        if (x.len < y.len) return true;
        if (y.len < x.len) return false;
        for (i = x.len - 1; i >= 0; i--) {
                if (x.num[i] < y.num[i]) return true;
                if (y.num[i] > x.num[i]) return false;
        }
        return false;
}
inline void print(INT x) {
        int i;
        for (i = x.len - 1; i >= 0; i--) write(x.num[i]);
        puts("");
}


int main() {
        
        read(n);
        read(a[0].l); read(a[0].r);
        for (i = 1; i <= n; i++) {
                read(a[i].l);
                read(a[i].r);     
        }
        sort(a+1,a+n+1,cmp);
        sum.len = 1; sum.num[0] = 1;
        
        for (i = 1; i <= n; i++) {
                multipy(sum,a[i-1].l);
                val = divide(sum,a[i].r);
                if (comp(ans,val)) ans = val;
        }
        print(ans);
        
        return 0;
    
}
