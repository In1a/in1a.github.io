using namespace std;
#define MAXT 1010
#define MAXS 100010
const int P = 1e6;
 
int i,j,T,A,S,B,sum,x,ans;
int s[MAXS];
int cnt[MAXT];
int f[2][MAXS];

int main() 
{
        
        scanf("%d%d%d%d",&T,&A,&S,&B);
        for (i = 1; i <= A; i++) 
        {
                scanf("%d",&x);
                cnt[x]++;
        }
        f[0][0] = 1;
        for (i = 1; i <= T; i++)
        {
                sum += cnt[i];
                s[0] = f[(i - 1) & 1][0];
                for (j = 1; j <= sum; j++) s[j] = (s[j - 1] + f[(i - 1) & 1][j]) % P;
                for (j = 0; j <= sum; j++)
                {
                        if (j <= cnt[i]) f[i & 1][j] = s[j];
                        else f[i & 1][j] = s[j] - s[j - cnt[i] - 1] + P;
                        f[i & 1][j] %= P;
                }         
        }
        for (i = S; i <= B; i++) ans = (ans + f[T & 1][i]) % P;
        printf("%d\n",ans);
        
        return 0;
    
}
