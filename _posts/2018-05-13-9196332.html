【算法】

           floyd求最小环

           输出路径的方法如下，对于i到j的最短路，我们记pre[i][j]表示j的上一步

           在进行松弛操作的时候更新pre即可

using namespace std;
#define MAXN 110
const int INF = 1e8;

int n,m;
int g[MAXN][MAXN],mp[MAXN][MAXN],pre[MAXN][MAXN];

inline void solve()
{
        int ans = INF;
        vector< int > res;
        for (int k = 1; k <= n; k++)
        {
                for (int i = 1; i < k; i++)
                {
                        for (int j = i + 1; j < k; j++)
                        {
                                if (g[i][j] + mp[j][k] + mp[k][i] < ans)
                                {
                                        ans = g[i][j] + mp[j][k] + mp[k][i];
                                        res.clear();
                                        int tmp = j;
                                        while (tmp != i)
                                        {
                                                res.push_back(tmp);
                                                tmp = pre[i][tmp];
                                        }    
                                        res.push_back(i);
                                        res.push_back(k);
                                }    
                        }    
                }    
                for (int i = 1; i <= n; i++)
                {
                        for (int j = 1; j <= n; j++)
                        {
                                if (g[i][k] + g[k][j] < g[i][j])
                                {
                                        g[i][j] = g[i][k] + g[k][j];
                                        pre[i][j] = pre[k][j];
                                }
                        }
                }
        }    
        if (ans == INF)
        {
                puts("No solution.");
                return;
        }
        for (int i = 0; i < res.size() - 1; i++) printf("%d ",res[i]);
        printf("%d\n",res[res.size()-1]);
}

int main() 
{
        
        scanf("%d%d",&n,&m);
        for (int i = 1; i <= n; i++)
        {
                for (int j = 1; j <= n; j++)
                {
                        g[i][j] = mp[i][j] = INF;
                        pre[i][j] = i;
                }
        }
        for (int i = 1; i <= m; i++)
        {
                int u,v,w;
                scanf("%d%d%d",&u,&v,&w);
                if (w < g[u][v])
                        g[u][v] = mp[u][v] = g[v][u] = mp[v][u] = w;
        }
        solve();
        
        return 0;
    
}
