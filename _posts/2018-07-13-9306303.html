using namespace std;
#define MAXN 8010

int i,n;
int a[MAXN],ans[MAXN],c[MAXN];

inline int lowbit(int x)
{
        return x & (-x);
}
inline void modify(int pos,int val)
{
        int i;
        for (i = pos; i <= n; i += lowbit(i)) c[i] += val; 
}
inline int query(int pos)
{
        int i;
        int ret = 0;
        for (i = pos; i; i -= lowbit(i)) ret += c[i];
        return ret;
}
inline int get(int k)
{
        int l = 1,r = n,mid;
        int ret;
        while (l <= r)
        {
                mid = (l + r) >> 1;
                if (query(mid) >= k)
                {
                        ret = mid;
                        r = mid - 1;
                } else l = mid + 1;
        }        
        return ret;
}

int main() 
{
        
        scanf("%d",&n);
        for (i = 2; i <= n; i++) scanf("%d",&a[i]);
        for (i = 1; i <= n; i++) c[i] = lowbit(i);
        for (i = n; i >= 1; i--)
        {
                ans[i] = get(a[i]+1);
                modify(ans[i],-1);        
        }
        for (i = 1; i <= n; i++) printf("%d\n",ans[i]);
        
        return 0;
    
}
