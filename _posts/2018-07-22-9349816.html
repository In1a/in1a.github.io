using namespace std;
#define MAXN 5010

int i,j,n,ans1,ans2;
int a[MAXN],f[MAXN],g[MAXN];

int main() 
{
        
        scanf("%d",&n);
        for (i = 1; i <= n; i++) 
        {
                scanf("%d",&a[i]);
                f[i] = g[i] = 1;
        }
        for (i = 1; i <= n; i++)
        {
                for (j = i - 1; j >= 1; j--)
                {
                        if (a[j] > a[i])
                        {
                                if (f[j] + 1 > f[i])
                                {
                                        f[i] = f[j] + 1;
                                        g[i] = g[j];    
                                }    else if (f[j] + 1 == f[i]) g[i] += g[j];
                        }    else if (a[i] == a[j])
                        {
                                if (f[i] == 1)
                                        g[i] = 0;
                                break;
                        }
                }    
        }
        for (i = 1; i <= n; i++)
        {
                if (f[i] > ans1)
                {
                        ans1 = f[i];
                        ans2 = g[i];    
                }    else if (f[i] == ans1) ans2 += g[i];
        }
        printf("%d %d\n",ans1,ans2);
        
        return 0;
    
}
