【算法】

          若A和B两头牛可以互相看见，那么说明中间的牛的高度都至少比它们少1，因此，我们可以引入一个差分数组c

          对于每组关系，我们将c[a+1]减一，c[b]加一

          最后，每头牛的最高高度就是 h + c的前缀和

using namespace std;
#define MAXN 100010

int i,a,b,n,p,h,m;
map< pair<int,int>,bool > mp;
int d[MAXN];

int main() 
{
        
        scanf("%d%d%d%d",&n,&p,&h,&m);
        for (i = 1; i <= m; i++)
        {
                scanf("%d%d",&a,&b);
                if (a > b) swap(a,b);
                if (mp[make_pair(a,b)]) continue;
                d[a+1]--; d[b]++;    
                mp[make_pair(a,b)] = true;
        }        
        for (i = 1; i <= n; i++)
        {
                d[i] += d[i-1];
                printf("%d\n",h+d[i]);    
        }
        
        return 0;
    
}
