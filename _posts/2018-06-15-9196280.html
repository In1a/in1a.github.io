using namespace std;
#define MAXN 30010
#define MAXM 150010
const int INF = 1e8;

struct Edge
{
        int to,w,nxt;
} e[MAXM];
int i,n,m,a,b,c,tot;
int dis[MAXN],head[MAXN];

inline void add(int u,int v,int w)
{
        tot++;
        e[tot] = (Edge){v,w,head[u]};
        head[u] = tot;
}
inline int spfa()
{
        int i,cur,v,w;
        stack<int> s;
        static bool instack[MAXN];
        static int cnt[MAXN];
        s.push(1);
        instack[1] = true;
        dis[1] = 0;        
        cnt[1] = 1;
        for (i = 2; i <= n; i++) dis[i] = INF; 
        while (!s.empty())
        {
                cur = s.top();
                s.pop();
                instack[cur] = false;
                for (i = head[cur]; i; i = e[i].nxt)
                {
                        v = e[i].to;
                        w = e[i].w;
                        if (dis[cur] + w < dis[v])
                        {
                                dis[v] = dis[cur] + w;
                                if (!instack[v])
                                {
                                        instack[v] = true;
                                        s.push(v);
                                        cnt[v]++;
                                        if (cnt[v] > n) return -1;
                                }
                        }
                }
        }
        return dis[n];
}

int main() 
{
        
        scanf("%d%d",&n,&m);
        for (i = 1; i <= m; i++)
        {
                scanf("%d%d%d",&a,&b,&c);
                add(a,b,c);        
        }
        printf("%d\n",spfa());
        
        return 0;
    
}
